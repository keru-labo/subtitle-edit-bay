cmake_minimum_required(VERSION 3.20)
project(subtitle_preview LANGUAGES CXX)

include(CTest)
if(BUILD_TESTING)
    add_executable(seek_gate_test ../../tests/preview/seek_gate_test.cpp)
    target_include_directories(seek_gate_test PRIVATE .)
    target_compile_features(seek_gate_test PRIVATE cxx_std_17)
    add_test(NAME seek_gate COMMAND seek_gate_test)
endif()

option(PREVIEW_D3D11_PROBE "Build the Windows FFmpeg texture probe" OFF)
if(PREVIEW_D3D11_PROBE)
    if(NOT WIN32)
        message(FATAL_ERROR "PREVIEW_D3D11_PROBE requires Windows")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
    add_executable(d3d11_decode_probe d3d11_decode_probe.cpp)
    target_compile_features(d3d11_decode_probe PRIVATE cxx_std_17)
    target_link_libraries(d3d11_decode_probe PRIVATE PkgConfig::FFMPEG d3d11)
endif()

option(PREVIEW_QT_PROBE "Build the Qt Multimedia sink probe" OFF)
if(PREVIEW_QT_PROBE)
    find_package(Qt6 REQUIRED COMPONENTS Multimedia)
    add_executable(qt_seek_probe qt_seek_probe.cpp)
    target_compile_features(qt_seek_probe PRIVATE cxx_std_17)
    target_link_libraries(qt_seek_probe PRIVATE Qt6::Multimedia)
endif()
